Bookstore API & Infinite Scroll Guide2React / Next.js implementationPlace the following client component in app/books/page.tsx. It follows Gutendex next URLs, avoids overlappingrequests, removes duplicate book IDs, and supports retry after an error. The observer loads another page whenthe sentinel approaches the viewport."use client";import { useCallback, useEffect, useRef, useState } from "react";type Book = {  id: number;  title: string;  authors: { name: string }[];  formats: Record<string, string>;};type BooksResponse = {  count: number;  next: string | null;  previous: string | null;  results: Book[];};const FIRST_PAGE = "https://gutendex.com/books/";export default function BooksPage() {  const [books, setBooks] = useState<Book[]>([]);  const [loading, setLoading] = useState(false);  const [error, setError] = useState<string | null>(null);  const [hasMore, setHasMore] = useState(true);  const nextUrlRef = useRef<string | null>(FIRST_PAGE);  const loadingRef = useRef(false);  const sentinelRef = useRef<HTMLDivElement | null>(null);  const loadMore = useCallback(async () => {    const url = nextUrlRef.current;    if (!url || loadingRef.current) return;    loadingRef.current = true;    setLoading(true);    setError(null);    try {      const response = await fetch(url);      if (!response.ok) {        throw new Error(`Unable to load books (${response.status}).`);      }      const data: BooksResponse = await response.json();      setBooks((previous) => {        const ids = new Set(previous.map((book) => book.id));        return [...previous,          ...data.results.filter((book) => !ids.has(book.id))];      });      nextUrlRef.current = data.next;      setHasMore(data.next !== null);    } catch (err) {      setError(err instanceof Error ? err.message : "Request failed.");    } finally {      loadingRef.current = false;      setLoading(false);    }  }, []);
